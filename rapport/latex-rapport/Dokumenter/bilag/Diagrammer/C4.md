            Rel(auth, prisma, "Henter Bruger oplysninger / refresh tokens")

        Component(apollo, "Apollo Controller", "Apollo", "Håndtere GraphQL anmodninger")
            Rel(apollo, prisma, "Henter/redigere de ønskede data")

            Rel(prisma, db, "Skriver/henter data")
